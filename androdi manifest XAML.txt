mainpage.xaml.cs


namespace AtikGozTemel;

public partial class MainPage : ContentPage
{
    public MainPage()
    {
        InitializeComponent();
        KonumAl();
    }

    private void Olc_Clicked(object sender, EventArgs e)
    {
        int mesafe = new Random().Next(5, 100);
        mesafeLabel.Text = $"Mesafe: {mesafe} cm";

        if (mesafe < 20)
            durumLabel.Text = "Durum: DOLU";
        else
            durumLabel.Text = "Durum: BOŞ";
    }

    private async void KonumAl()
    {
        try
        {
            var location = await Geolocation.GetLastKnownLocationAsync();

            if (location != null)
            {
                konumLabel.Text =
                    $"Konum: {location.Latitude:F4}, {location.Longitude:F4}";
            }
            else
            {
                konumLabel.Text = "Konum alınamadı";
            }
        }
        catch
        {
            konumLabel.Text = "Konum izni reddedildi";
        }
    }
}


mauiprogram.cs

using Microsoft.Extensions.Logging;

namespace AtikGozTemel
{
    public static class MauiProgram
    {
        public static MauiApp CreateMauiApp()
        {
            var builder = MauiApp.CreateBuilder();
            builder
                .UseMauiApp<App>()
                .ConfigureFonts(fonts =>
                {
                    fonts.AddFont("OpenSans-Regular.ttf", "OpenSansRegular");
                    fonts.AddFont("OpenSans-Semibold.ttf", "OpenSansSemibold");
                });

#if DEBUG
    		builder.Logging.AddDebug();
#endif

            return builder.Build();
        }
    }
}
